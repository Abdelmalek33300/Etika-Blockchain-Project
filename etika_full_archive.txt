/// Fonction pour générer le message standard qui sera signé pour les transactions PoP
fn generate_pop_message(transaction: &PoPTransaction<T::AccountId, T::BlockNumber>) -> Vec<u8> {
    // Utilisation de l'encodage SCALE pour garantir la cohérence
    (
        transaction.id,
        transaction.consumer.clone(),
        transaction.merchant.clone(),
        transaction.suppliers.clone(),
        transaction.standard_amount,
        transaction.tokens_exchanged,
        transaction.savings_generated,
        transaction.timestamp,
        transaction.receipt_hash
    ).encode()
}

/// Validation de la transaction PoP avec vérification cryptographique
fn validate_pop_transaction(transaction: &PoPTransaction<T::AccountId, T::BlockNumber>) -> Result<(), Error<T>> {
    // Vérifier que la transaction contient au moins deux signatures (consommateur et marchand)
    ensure!(transaction.signatures.len() >= 2, Error::<T>::InsufficientSignatures);
    
    // Générer le message qui a été signé
    let message = Self::generate_pop_message(transaction);
    
    // Vérifier chaque signature
    for (signer, signature) in &transaction.signatures {
        // Convertir AccountId en clé publique au format attendu par sr25519_verify
        let account_id_bytes = signer.encode();
        
        // Vérifier si la longueur des bytes correspond à celle d'une clé publique sr25519
        ensure!(account_id_bytes.len() == 32, Error::<T>::InvalidSignerFormat);
        
        // Convertir la signature en tableau de 64 bytes
        let sig_bytes: [u8; 64] = match signature.try_into() {
            Ok(array) => array,
            Err(_) => return Err(Error::<T>::InvalidSignatureFormat),
        };
        
        // Vérifier la signature avec sr25519_verify
        let is_valid = sp_io::crypto::sr25519_verify(
            &sig_bytes,
            &message,
            &account_id_bytes.try_into().unwrap_or([0u8; 32])
        );
        
        // Si la signature n'est pas valide, retourner une erreur
        if !is_valid {
            return Err(Error::<T>::InvalidSignature);
        }
    }
    
    // Vérifier que le consommateur est bien un signataire
    let consumer_signed = transaction.signatures.iter()
        .any(|(signer, _)| signer == &transaction.consumer);
    
    ensure!(consumer_signed, Error::<T>::ConsumerSignatureRequired);
    
    // Vérifier que le marchand est bien un signataire
    let merchant_signed = transaction.signatures.iter()
        .any(|(signer, _)| signer == &transaction.merchant);
    
    ensure!(merchant_signed, Error::<T>::MerchantSignatureRequired);
    
    // Toutes les vérifications ont réussi
    Ok(())
}

// Configuration pour la durée d'un bloc
pub trait Config: frame_system::Config + Sized {
    // Autres paramètres existants...
    
    /// Durée moyenne d'un bloc en millisecondes
    type BlockDuration: Get<u64>;
    
    /// Nombre d'avertissements requis avant une sanction
    type WarningsBeforeSanction: Get<u32>;
    
    /// Durée maximale d'attente pour la validation par les administrateurs (en blocs)
    type AdminValidationTimeout: Get<Self::BlockNumber>;
    
    /// Allocation par défaut de tokens pour les nouvelles enchères
    type DefaultAuctionTokenAllocation: Get<Balance>;
}

// Dans parameter_types!
parameter_types! {
    // Autres paramètres existants...
    
    // Durée moyenne d'un bloc (6 secondes = 6000 ms)
    pub const BlockDuration: u64 = 6000;
    
    // Nombre d'avertissements avant sanction (généralement 1-3)
    pub const WarningsBeforeSanction: u32 = 2;
    
    // Environ 1 jour à 6s par bloc
    pub const AdminValidationTimeout: u32 = 14400;
    
    // Allocation par défaut de tokens pour les enchères
    pub const DefaultAuctionTokenAllocation: Balance = 100_000;
}

impl Config for Runtime {
    // Implémentation existante...
    
    type BlockDuration = BlockDuration;
    type WarningsBeforeSanction = WarningsBeforeSanction;
    type AdminValidationTimeout = AdminValidationTimeout;
    type DefaultAuctionTokenAllocation = DefaultAuctionTokenAllocation;
}

// Fonction pour enregistrer l'adoption basée sur une transaction PoP
fn register_pop_adoption_evidence(transaction: &PoPTransaction<T::AccountId, T::BlockNumber>) -> DispatchResult {
    let consumer = &transaction.consumer;
    let merchant = &transaction.merchant;
    
    // Récupérer les catégories sponsorisées
    let categories = Self::get_all_sponsored_categories();
    
    // Pour chaque catégorie où le marchand est sponsor, enregistrer l'adoption
    for category in categories {
        if let Some(sponsor_id) = <CategorySponsors<T>>::get(&category) {
            if sponsor_id == *merchant {
                // Enregistrer l'adoption
                <AdoptionStatus<T>>::insert((consumer.clone(), category.clone()), true);
                
                // Estimer le numéro de bloc à partir du timestamp
                // Note: C'est une approximation, documentée comme telle
                let estimated_block = T::BlockNumber::from(
                    (transaction.timestamp / T::BlockDuration::get()) as u32
                );
                
                // Mettre à jour la dernière interaction
                <LastSponsorInteraction<T>>::insert(
                    (consumer.clone(), category.clone()),
                    estimated_block
                );
                
                // Mettre à jour le score d'engagement
                <SponsorEngagementScore<T>>::mutate(
                    (consumer.clone(), category.clone()),
                    |score| *score = score.saturating_add(1)
                );
                
                // Émettre un événement
                Self::deposit_event(RawEvent::SponsorAdoptionRegistered(
                    consumer.clone(),
                    category.clone(),
                    sponsor_id,
                    estimated_block
                ));
            }
        }
    }
    
    Ok(())
}

// Fonction interne pour créer une enchère
fn create_auction_internal(
    category: Vec<u8>,
    territory_id: Vec<u8>,
    start_block: T::BlockNumber,
    end_block: T::BlockNumber,
    starting_price: Balance,
    min_bid_increment: Balance,
    retry_count: u32
) -> Result<[u8; 32], Error<T>> {
    // Vérifier que la catégorie existe
    ensure!(
        <Categories<T>>::contains_key(&category),
        Error::<T>::CategoryNotFound
    );
    
    // Vérifier que le territoire existe
    ensure!(
        <Territories<T>>::contains_key(&territory_id) ||
        territory_id == b"default".to_vec(),
        Error::<T>::TerritoryNotFound
    );
    
    // Vérifier que la période d'enchère est valide
    ensure!(end_block > start_block, Error::<T>::InvalidAuctionPeriod);
    
    // Vérifier que le prix de départ est valide
    ensure!(
        starting_price >= T::MinimumAuctionStartingPrice::get(),
        Error::<T>::InvalidStartingPrice
    );
    
    // Vérifier qu'aucune enchère active n'existe déjà pour cette catégorie
    let last_auction_id = <CategoryLastAuction<T>>::get(&category);
    
    if let Some(last_auction) = <Auctions<T>>::get(last_auction_id) {
        if last_auction.status == AuctionStatus::Active || 
           last_auction.status == AuctionStatus::Pending {
            return Err(Error::<T>::CategoryAlreadyHasActiveAuction);
        }
    }
    
    // Générer un nouvel ID pour l'enchère
    let auction_id = Self::generate_auction_id();
    
    // Créer la structure d'enchère
    let auction = AuctionData {
        id: auction_id,
        category: category.clone(),
        territory_id: territory_id.clone(),
        start_block,
        end_block,
        starting_price,
        min_bid_increment,
        status: AuctionStatus::Pending,
        bid_history: Vec::new(),
        validating_admins: Vec::new(),
        admin_validation: false,
        retry_count,
        participants: Vec::new(),
        tokens_allocated: T::DefaultAuctionTokenAllocation::get(),
        tokens_distributed: 0,
        finalization_error: None,
        finalized_at: None,
    };
    
    // Stocker l'enchère
    <Auctions<T>>::insert(auction_id, auction);
    
    // Mettre à jour les métadonnées de la catégorie
    <Categories<T>>::mutate(&category, |metadata| {
        if let Some(meta) = metadata {
            meta.auction_count += 1;
        }
    });
    
    // Mettre à jour l'ID de la dernière enchère pour cette catégorie
    <CategoryLastAuction<T>>::insert(category.clone(), auction_id);
    
    // Émettre un événement
    Self::deposit_event(RawEvent::AuctionCreated(
        auction_id,
        category,
        territory_id,
        starting_price
    ));
    
    Ok(auction_id)
}
